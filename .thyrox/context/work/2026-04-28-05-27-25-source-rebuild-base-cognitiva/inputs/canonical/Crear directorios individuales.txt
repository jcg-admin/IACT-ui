# Crear directorios individuales
mkdir -p /tmp/iact_regeneracion/fundacionales
mkdir -p /tmp/iact_regeneracion/partes
mkdir -p /tmp/iact_regeneracion/templates
mkdir -p /tmp/iact_regeneracion/ejemplos
mkdir -p /tmp/iact_regeneracion/indices

echo "✅ Directorios creados:"
tree -L 2 /tmp/iact_regeneracion/ 2>/dev/null || ls -R /tmp/iact_regeneracion/
